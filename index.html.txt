        let selectedItems = new Set();
        let currentTheme = 'sunset';

        // Initialize Checklist UI
        function renderChecklist() {
            const container = document.getElementById('checklist-container');
            container.innerHTML = '';

            categories.forEach((cat, catIndex) => {
                let catDiv = document.createElement('div');
                catDiv.className = 'bg-white/90 backdrop-blur-md rounded-2xl p-5 shadow-md border border-white';
                
                let titleHTML = `<h2 class="text-xl font-bold mb-4 text-rose-600 border-b pb-2 flex items-center gap-2">${cat.title}</h2>`;
                
                let gridHTML = `<div class="grid grid-cols-1 sm:grid-cols-2 gap-3">`;
                cat.items.forEach((item, itemIndex) => {
                    let id = `cat-${catIndex}-item-${itemIndex}`;
                    gridHTML += `
                        <div onclick="toggleItem('${item}')" id="${id}" class="checkbox-card cursor-pointer p-3 rounded-xl border-2 border-gray-200 hover:border-rose-400 transition flex items-center gap-3 bg-gray-50/50 select-none">
                            <div class="w-6 h-6 rounded-lg border-2 border-gray-400 flex items-center justify-center text-white check-box-icon transition">
                                <i class="fa-solid fa-check text-xs opacity-0"></i>
                            </div>
                            <span class="text-sm font-medium text-gray-700">${item}</span>
                        </div>
                    `;
                });
                gridHTML += `</div>`;
                
                catDiv.innerHTML = titleHTML + gridHTML;
                container.appendChild(catDiv);
            });
        }

        // Toggle Item Selection
        function toggleItem(itemText) {
            if (selectedItems.has(itemText)) {
                selectedItems.delete(itemText);
            } else {
                selectedItems.add(itemText);
            }
            updateUI();
        }

        // Update Score and Visuals
        function updateUI() {
            let totalItems = 0;
            categories.forEach(c => totalItems += c.items.length);
            
            let count = selectedItems.size;
            let percentage = Math.round((count / totalItems) * 100);

            document.getElementById('score-text').innerText = percentage + '%';
            document.getElementById('progress-bar').style.width = percentage + '%';

            // Highlight Cards in DOM
            categories.forEach((cat, catIndex) => {
                cat.items.forEach((item, itemIndex) => {
                    let id = `cat-${catIndex}-item-${itemIndex}`;
                    let el = document.getElementById(id);
                    if (el) {
                        let iconBox = el.querySelector('.check-box-icon');
                        let checkIcon = iconBox.querySelector('i');
                        if (selectedItems.has(item)) {
                            el.classList.add('border-rose-500', 'bg-rose-50/80');
                            el.classList.remove('border-gray-200', 'bg-gray-50/50');
                            iconBox.classList.add('bg-rose-500', 'border-rose-500');
                            checkIcon.classList.remove('opacity-0');
                        } else {
                            el.classList.remove('border-rose-500', 'bg-rose-50/80');
                            el.classList.add('border-gray-200', 'bg-gray-50/50');
                            iconBox.classList.remove('bg-rose-500', 'border-rose-500');
                            checkIcon.classList.add('opacity-0');
                        }
                    }
                });
            });
        }

        // Modal Open & Badge Calculation
        function openCardModal() {
            let totalItems = 0;
            categories.forEach(c => totalItems += c.items.length);
            let percentage = Math.round((selectedItems.size / totalItems) * 100);

            document.getElementById('card-score-badge').innerText = percentage + '%';
            
            let badgeTitle = "";
            let tagline = "";

            if (percentage >= 80) {
                badgeTitle = "🏆 ৯০ দশকের গোল্ডেন চাইল্ড";
                tagline = "আপনি খাঁটি শৈশব উপভোগ করেছেন! বর্তমান যুগের বাচ্চারা এই আনন্দের ১০%ও পাবে না।";
            } else if (percentage >= 50) {
                badgeTitle = "🥈 নস্টালজিক মেমোরি মাস্টার";
                tagline = "আপনার শৈশব বেশ রঙিন ছিল, অনেক দারুণ স্মৃতি আজও মনকে ছুঁয়ে যায়।";
            } else if (percentage >= 20) {
                badgeTitle = "🥉 অর্ধেক নোস্টালজিক";
                tagline = "স্মৃতির কিছু পাতা মিস হয়ে গেছে, তবুও মন্দ নয়!";
            } else {
                badgeTitle = "🌱 ডিজিটাল যুগের বাচ্চা";
                tagline = "আপনি মনে হয় একটু বেশিই স্মার্ট যুগে বড় হয়েছেন!";
            }

            document.getElementById('card-badge-title').innerText = badgeTitle;
            document.getElementById('card-tagline').innerText = tagline;

            // Render Highlights List
            let highlightsUl = document.getElementById('card-highlights');
            highlightsUl.innerHTML = '';
            let arr = Array.from(selectedItems).slice(0, 4);
            if (arr.length === 0) {
                highlightsUl.innerHTML = `<li class="italic opacity-70">কোনো স্মৃতি সিলেক্ট করা হয়নি!</li>`;
            } else {
                arr.forEach(item => {
                    highlightsUl.innerHTML += `<li>${item}</li>`;
                });
            }

            // Generate QR Code
            document.getElementById('qrcode').innerHTML = "";
            new QRCode(document.getElementById("qrcode"), {
                text: window.location.href,
                width: 44,
                height: 44,
                colorDark : "#000000",
                colorLight : "#ffffff",
                correctLevel : QRCode.CorrectLevel.H
            });
            
            document.getElementById('site-url-text').innerText = window.location.host || "shishobnama.vercel.app";

            document.getElementById('modal').classList.remove('hidden');
        }

        function closeCardModal() {
            document.getElementById('modal').classList.add('hidden');
        }

        // Theme Changer for Card
        function setTheme(theme) {
            currentTheme = theme;
            let card = document.getElementById('capture-card');
            card.className = "rounded-2xl p-6 text-white shadow-xl transition-all duration-300 relative overflow-hidden mb-6";
            
            if (theme === 'sunset') {
                card.classList.add('bg-gradient-to-br', 'from-amber-500', 'via-rose-500', 'to-purple-600');
            } else if (theme === 'mint') {
                card.classList.add('bg-gradient-to-br', 'from-teal-400', 'via-emerald-500', 'to-cyan-700');
            } else if (theme === 'vintage') {
                card.classList.add('bg-gradient-to-br', 'from-amber-700', 'via-yellow-700', 'to-stone-800');
            } else if (theme === 'purple') {
                card.classList.add('bg-gradient-to-br', 'from-purple-600', 'via-indigo-700', 'to-slate-900');
            }
        }

        // Download Card as PNG
        function downloadCard() {
            let cardElement = document.getElementById('capture-card');
            html2canvas(cardElement, { scale: 3, useCORS: true }).then(canvas => {
                let link = document.createElement('a');
                link.download = 'my-childhood-score.png';
                link.href = canvas.toDataURL('image/png');
                link.click();
            });
        }

        // Run on load
        window.onload = renderChecklist;
